techman-js

TypeScript-библиотека для управления промышленным коботом Techman Robot TM5-700 по протоколу TCP/IP через интерфейс Listen Node. Реализует самостоятельную сборку бинарных пакетов, расчёт контрольных сумм (XOR), валидацию длины данных и парсинг ответов робота — без использования сторонних SDK.

import { IConnection, ParsedTMSCT } from '../types/index.js';
import { TMSCTBuilder } from '../protocol/builders/TMSCTBuilder.js';

export class TMScriptClient {
  constructor(private connection: IConnection) {}
  
  async ptp(target: number[], format: 'JPP' | 'CPP', 
            speed = 5, accelMs = 200, blend = 0) {
    const packet = this.motion.PTP(target, format, speed, accelMs, blend);
    return this.connection.execute<ParsedTMSCT>(packet);
  }
}

Пример: packet-builder.ts

export class PacketBuilder implements IPacketBuilder {
  build(header: TMHeader, payload: string): BuiltPacket {
    const generatedId = this.generateId();
    const data = `${generatedId},${payload}`;
    const length = this.calculateByteLength(data);
    const content = `${header},${length},${data},`;
    const checksum = this.calculateChecksum(content);
    return { raw: `$${content}*${checksum}\r\n`, id: generatedId };
  }
  
  private calculateChecksum(data: string): string {
    const buffer = Buffer.from(data, 'utf8');
    let checksum = 0;
    for (let i = 0; i < buffer.length; i++) {
      checksum ^= buffer[i];
    }
    return checksum.toString(16).toUpperCase().padStart(2, '0');
  }
}

Пример: textMapper.ts

export function textToCellMap(text: string, scale: number, 
                               font: FontData, settings: RobotSettings) {
  const { x: fieldX, y: fieldY } = settings.fieldSize;
  const scaledWidth = font.charWidth * scale;
  const scaledHeight = font.charHeight * scale;
  const cellCountX = Math.floor(fieldX / scaledWidth);
  const cellCountY = Math.floor(fieldY / scaledHeight);
  
  const words = text.split(/\s+/).filter(w => w.length > 0);
  const result: CharCell[] = [];
  let currentX = 0;
  let currentY = cellCountY - 1;
  
  for (const word of words) {
    if (currentX + word.length > cellCountX) {
      currentY--; currentX = 0;
    }
    for (const char of word) {
      result.push({
        char, x_index: currentX, y_index: currentY,
        abs_x: currentX * scaledWidth,
        abs_y: currentY * scaledHeight
      });
      currentX++;
    }
  }
  return result;
}
